pipeline {
    agent any

    environment {
        IMAGE_NAME = 'proctomtechdocker/weeky-travel-portal'
        IMAGE_TAG = 'latest'
        PORT_MAPPING = '9096:9096'
        CONTAINER_NAME = 'weeky-travel-portal'
        DOCKERFILE_PATH = 'Dockerfile'
        HOST_UPLOAD_DIR = '/opt/upload-files'
        CONTAINER_UPLOAD_DIR = '/opt/upload-files'
    }

    stages {
        stage('Checkout Code') {
            steps {
                git branch: 'main', credentialsId: 'github-cred', url: 'https://github.com/upendra-rai/weeky-travel-portal.git'
            }
        }

        stage('Verify Workspace Contents') {
            steps {
                echo "📁 Listing root directory files:"
                sh "ls -la"
                echo "📁 Checking for deploy directory:"
                sh "ls -la deploy || echo 'deploy directory does not exist'"
            }
        }

        stage('Build Docker Image') {
            steps {
                script {
                    echo "🔨 Building Docker image..."
                    sh "docker rmi -f ${IMAGE_NAME}:${IMAGE_TAG} || true"
                    sh "docker build --progress=plain -t ${IMAGE_NAME}:${IMAGE_TAG} -f ${DOCKERFILE_PATH} ."
                }
            }
        }

        stage('Stop and Remove Existing Container') {
            steps {
                script {
                    echo "🧹 Cleaning up old container if exists..."
                    sh "docker stop ${CONTAINER_NAME} || true"
                    sh "docker rm ${CONTAINER_NAME} || true"
                }
            }
        }

        stage('Run New Container') {
            steps {
                script {
                    echo "🚀 Running container ${CONTAINER_NAME} on port ${PORT_MAPPING} with persistent volume..."
                    sh """
                        docker run -d \
                        -p ${PORT_MAPPING} \
                        -v ${HOST_UPLOAD_DIR}:${CONTAINER_UPLOAD_DIR} \
                        --name ${CONTAINER_NAME} \
                        ${IMAGE_NAME}:${IMAGE_TAG}
                    """
                }
            }
        }
    }

    post {
        success {
            echo "✅ Deployment successful! Container is running on http://<your-server-ip>:9096"
        }
        failure {
            echo "❌ Deployment failed. Check Jenkins logs for troubleshooting."
        }
    }
}
