pipeline {
    agent any

    environment {
        APP_NAME = "weeky-travel-connect-partner-app"
        GIT_URL = "git@github.com:upendra-rai/weeky-travel-connect-partner-app.git"
        BRANCH = "main" // or 'stagging', etc.
        DEPLOY_DIR = "/var/www/weeky-travel-connect-partner-app"
        NODE_ENV = "production"
        SERVICE_NAME = "weeky-travel-connect-partner-app.service"
        CREDENTIALS_ID = "github-cred"
        BUILD_ARTIFACT_DIR = "build_artifacts"
    }

    stages {

        stage('Cleanup Workspace') {
            steps {
                echo "🧹 Cleaning workspace..."
                deleteDir()
            }
        }

        stage('Checkout Code') {
            steps {
                echo "📦 Checking out branch: ${BRANCH}"
                git branch: "${BRANCH}", credentialsId: "${CREDENTIALS_ID}", url: "${GIT_URL}"
            }
        }

        stage('Install & Build') {
            steps {
                echo "🛠️ Installing dependencies and building Next.js app..."
                sh '''
                    # Use clean install to ensure consistent builds
                    npm ci --legacy-peer-deps || npm install --legacy-peer-deps
                    
                    # Build production-ready Next.js app
                    npm run build

                    # Create a lightweight artifact directory for deployment
                    rm -rf ${BUILD_ARTIFACT_DIR}
                    mkdir -p ${BUILD_ARTIFACT_DIR}

                    # Copy only required production files
                    cp -r .next package.json next.config.js public ${BUILD_ARTIFACT_DIR}/ 2>/dev/null || true
                    cp -r node_modules ${BUILD_ARTIFACT_DIR}/node_modules
                '''
            }
        }

        stage('Deploy to Server') {
            steps {
                echo "🚀 Deploying app to ${DEPLOY_DIR}"
                sh '''
                    # Ensure target directory exists
                    sudo mkdir -p ${DEPLOY_DIR}

                    # Sync built files only
                    sudo rsync -avz --delete \
                        --exclude=".git" \
                        ${BUILD_ARTIFACT_DIR}/ ${DEPLOY_DIR}/

                    # Set proper permissions
                    sudo chown -R www-data:www-data ${DEPLOY_DIR}
                '''
            }
        }

        stage('Restart Service') {
            steps {
                echo "🔁 Restarting systemd service: ${SERVICE_NAME}"
                sh '''
                    sudo systemctl daemon-reload
                    sudo systemctl restart ${SERVICE_NAME}
                    sudo systemctl enable ${SERVICE_NAME}
                    sudo systemctl status ${SERVICE_NAME} --no-pager || true
                '''
            }
        }
    }

    post {
        success {
            echo "✅ Deployment successful!"
            echo "🌐 App is running under systemd service: ${SERVICE_NAME}"
        }
        failure {
            echo "❌ Deployment failed. Check logs using:"
            echo "👉 sudo journalctl -u ${SERVICE_NAME} -n 50"
        }
    }
}
