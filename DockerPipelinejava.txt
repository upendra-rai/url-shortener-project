pipeline {
    agent any

    environment {
        PROJECT_NAME = 'weeky-travel-portal'
        GIT_URL = 'https://github.com/upendra-rai/weeky-travel-portal.git'
        BRANCH = 'stagging'
        DEPLOY_DIR = '/root/app/weeky-travel-portal'
        SERVICE_NAME = 'weeky-travel-portal.service'
        JAR_NAME = 'weeky-travel-portal-0.0.1-SNAPSHOT.jar'
        JAVA_HOME = '/usr/lib/jvm/java-17-openjdk-amd64'
        PATH = "${JAVA_HOME}/bin:$PATH"
    }

    stages {
        stage('Checkout Code') {
            steps {
                git branch: "${BRANCH}", credentialsId: 'github-cred', url: "${GIT_URL}"
            }
        }

        stage('Verify Workspace Contents') {
            steps {
                echo "📁 Listing project files:"
                sh "ls -la"
            }
        }

        stage('Build JAR') {
            steps {
                script {
                    echo "🔨 Building JAR file..."
                    sh 'mvn clean package -DskipTests'
                }
            }
        }

        stage('Deploy to Linux System') {
            steps {
                script {
                    echo "🚀 Deploying application to ${DEPLOY_DIR}"

                    // Stop service if running
                    sh """
                        if systemctl is-active --quiet ${SERVICE_NAME}; then
                            echo "Stopping existing service..."
                            sudo systemctl stop ${SERVICE_NAME}
                        else
                            echo "No running service found."
                        fi
                    """

                    // Copy new JAR
                    sh """
                        sudo mkdir -p ${DEPLOY_DIR}
                        sudo cp target/${JAR_NAME} ${DEPLOY_DIR}/${JAR_NAME}
                        sudo chown root:root ${DEPLOY_DIR}/${JAR_NAME}
                    """

                    // Restart service
                    sh """
                        echo "Restarting ${SERVICE_NAME}..."
                        sudo systemctl daemon-reload
                        sudo systemctl start ${SERVICE_NAME}
                        sudo systemctl enable ${SERVICE_NAME}
                    """
                }
            }
        }
    }

    post {
        success {
            echo "✅ Deployment successful! App running at http://62.72.57.219:9096"
        }
        failure {
            echo "❌ Deployment failed. Check logs for details."
        }
    }
}
